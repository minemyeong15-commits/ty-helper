Firebase는 request.auth와 인증 토큰 정보를 Rules에서 사용할 수 있다고 안내한다. // ============================================================ // 통영고등학교 도우미 - Firestore Security Rules // ============================================================ // // 적용 위치: // Firebase Console // → Firestore Database // → Rules // → 아래 전체를 붙여넣기 // → 게시(Publish) // // 데이터 구조: // qna // studentQna // studentQna/{docId}/comments // lostItems // lostItems/{docId}/comments // uniforms // uniformApplicants // // 권한: // - 비로그인 사용자: 공개 데이터 읽기 / 질문 / 분실물 / 댓글 / 교복 신청 // - 관리자(Firebase Authentication 로그인): 전체 관리 // // 주의: // 현재 프로젝트는 별도 서버 없이 Firebase Web SDK에서 직접 Firestore를 사용하므로 // 실제 보안은 반드시 이 Rules에서 강제되어야 한다. // ============================================================ rules_version = '2'; } // 문자열 길이 검사 function isShortString(value, maxLen) { return value is string && value.size() > 0 && value.size() <= maxLen; } // createdAt 검사 function isValidCreatedAt(value) { return value is number && value >= 0; } // ========================================================== // 1. 선생님 질문 게시판 // 컬렉션: qna // ========================================================== match /qna/{docId} { // 학생 / 방문자 모두 질문 목록 열람 가능 allow read: if true; // 학생 질문 생성 // // 실제 HTML 코드: // text // hashtags // createdAt // allow create: if request.resource.data.keys().hasAll([ 'text', 'hashtags', 'createdAt' ]) && request.resource.data.keys().hasOnly([ 'text', 'hashtags', 'createdAt' ]) && isShortString( request.resource.data.text, 500 ) && request.resource.data.hashtags is list && isValidCreatedAt( request.resource.data.createdAt ); // 질문 수정 / 삭제는 관리자만 allow update, delete: if isAdmin(); // -------------------------------------------------------- // 학생 질문 댓글 // 경로: // studentQna/{docId}/comments/{commentId} // -------------------------------------------------------- match /comments/{commentId} { // 댓글 보기 allow read: if true; // 댓글 수정 / 삭제는 관리자만 allow update, delete: if isAdmin(); } } // ========================================================== // 4. 교복 재고 // 컬렉션: uniforms // ========================================================== // // 현재 실제 데이터 구조: // // uniforms/{docId} // // { // category: "하복", // name: "하복 상의", // price: "1,000원", // sizes: { // "85": 3, // "90": 5, // "95": 2 // } // } // // 중요: // 현재 HTML은 qty가 아니라 sizes Map을 사용한다. // ========================================================== match /uniforms/{docId} { // 재고는 누구나 볼 수 있음 allow read: if true; // 교복 신청 allow create: if request.resource.data.keys().hasAll([ 'item', 'size', 'studentId', 'name', 'date', 'createdAt' ]) && request.resource.data.keys().hasOnly([ 'item', 'size', 'studentId', 'name', 'date', 'createdAt' ]) && isShortString( request.resource.data.item, 100 ) && isShortString( request.resource.data.size, 10 ) && isShortString( request.resource.data.studentId, 20 ) && isShortString( request.resource.data.name, 30 ) && isShortString( request.resource.data.date, 30 ) && isValidCreatedAt( request.resource.data.createdAt ); // 신청 내역 수정 / 삭제는 관리자만 allow update, delete: if isAdmin(); } // ========================================================== // 6. 위에서 선언하지 않은 모든 경로는 차단 // ========================================================== match /{document=**} { allow read, write: if false; } } } 이 규칙으로 바뀌는 부분 현재 코드 기준으로 보면 기존 Rules에서 문제가 되던 qna.hashtags 누락, studentQna 자체 누락, 두 종류의 comments 누락, uniformApplicants.size 누락을 모두 해결했다. 실제 HTML에서도 학생 질문 댓글과 분실물 댓글을 각각 하위 컬렉션으로 사용하고 있다. 교복도 기존 qty 방식이 아니라 현재 코드의 sizes Map 구조에 맞췄다. 코드가 실제로 sizes[size] = newQty 형태로 저장하고 있다. 또한 신청자 문서에 실제로 size가 들어가므로 Rules에도 size를 포함했다. ⚠️ 한 가지는 꼭 알고 있어야 해 현재 구조에서 가장 약한 부분은 교복 재고 sizes Map이야. 학생 신청 코드는 현재 한 문서의 sizes 전체를 읽은 다음 특정 사이즈를 -1해서 다시 전체 Map을 저장한다. Firestore Rules만으로는 동적으로 입력되는 임의의 사이즈 키 각각에 대해 기존 값 → 정확히 1 감소를 완벽하게 검증하기 어렵다. Firebase 공식 문서도 Map/List 내부 모든 구성원의 타입을 일괄 검사하는 기능에는 제한이 있다고 명시한다. 따라서 실제 서비스 수준의 보안까지 생각하면 다음 단계에서는 uniforms의 사이즈를 별도 문서/하위 컬렉션으로 분리하거나 서버 측 처리로 바꾸는 것이 가장 안전하다. 또한 Firebase는 App Check를 함께 사용하는 것도 권장하고 있다. 그리고 이 Rules를 적용한 뒤에는 ** Firebase 콘솔의 Rules Playground/에뮬레이터에서 최소한 학생 질문 등록, 댓글 등록, 분실물 등록, 교복 신청, 관리자 삭제를 각각 테스트하는 것**이 좋다.
